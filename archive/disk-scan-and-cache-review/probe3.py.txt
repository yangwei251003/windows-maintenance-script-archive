import os, datetime
def fmt(n):
    for u in ['B','KB','MB','GB','TB']:
        if n<1024: return f"{n:.1f} {u}"
        n/=1024
    return f"{n:.1f} TB"
def dsize(p):
    t=0
    for r,d,fs in os.walk(p):
        for f in fs:
            try: t+=os.path.getsize(os.path.join(r,f))
            except: pass
    return t
def drill(root, label, n=30):
    print(f"\n===== {label} =====\n{root}")
    if not os.path.isdir(root): print("  [不存在]"); return
    es=[]
    for name in os.listdir(root):
        p=os.path.join(root,name)
        try: mt=datetime.datetime.fromtimestamp(os.path.getmtime(p)).strftime("%Y-%m-%d")
        except: mt="?"
        if os.path.isdir(p): es.append((dsize(p),name,mt,"dir"))
        else:
            try: s=os.path.getsize(p)
            except: s=0
            es.append((s,name,mt,"file"))
    es.sort(reverse=True)
    for s,name,mt,t in es[:n]:
        print(f"  {fmt(s):>10}  {mt}  {t}  {name}")

import os, shutil
root=r"<LOCAL_USER_HOME>\.npm-cache"
def fmt(n):
    for u in ['B','KB','MB','GB','TB']:
        if n<1024: return f"{n:.1f} {u}"
        n/=1024
    return f"{n:.1f} TB"
def dsize(p):
    t=0
    for r,d,fs in os.walk(p):
        for f in fs:
            try: t+=os.path.getsize(os.path.join(r,f))
            except: pass
    return t
total=0
print("===== 清理 .npm-cache =====")
for name in os.listdir(root):
    p=os.path.join(root,name)
    s=dsize(p) if os.path.isdir(p) else (os.path.getsize(p) if os.path.isfile(p) else 0)
    try:
        if os.path.isdir(p) and not os.path.islink(p):
            shutil.rmtree(p, ignore_errors=True)
        else:
            os.remove(p)
        total+=s
        print(f"  [删除] {fmt(s):>10}  {name}")
    except Exception as e:
        print(f"  [跳过] {name}: {e}")
print(f"\n===== .npm-cache 释放: {fmt(total)} =====")
print("清理后剩余:", os.listdir(root) if os.path.isdir(root) else "目录不存在")

import os
def dir_size(path, max_files=500000):
    total=0; count=0
    try:
        for root,dirs,files in os.walk(path):
            for f in files:
                try: total+=os.path.getsize(os.path.join(root,f)); count+=1
                except: pass
                if count>=max_files: return total,count
    except: pass
    return total,count
def fmt(n):
    for u in ['B','KB','MB','GB','TB']:
        if n<1024: return f"{n:.1f} {u}"
        n/=1024
    return f"{n:.1f} TB"

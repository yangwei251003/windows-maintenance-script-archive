import os, datetime
def fmt(n):
    for u in ['B','KB','MB','GB','TB']:
        if n<1024: return f"{n:.1f} {u}"
        n/=1024
    return f"{n:.1f} TB"
def ls(root,label,n=12):
    print(f"\n===== {label} =====\n{root}")
    if not os.path.isdir(root): print("  [不存在]"); return
    for name in sorted(os.listdir(root))[:n]:
        p=os.path.join(root,name)
        try: s=os.path.getsize(p) if os.path.isfile(p) else 0
        except: s=0
        try: mt=datetime.datetime.fromtimestamp(os.path.getmtime(p)).strftime("%Y-%m-%d")
        except: mt="?"
        typ="dir" if os.path.isdir(p) else "file"
        print(f"  {fmt(s):>8}  {mt}  {typ}  {name}")

# -*- coding: utf-8 -*-
"""导出可编辑的 Excel 软件清单（带'我的决定'下拉列）"""
import sys, os
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import build_report as BR  # 复用扫描与分类逻辑

from openpyxl import Workbook
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter
from openpyxl.worksheet.datavalidation import DataValidation

ADV = {
    'keep':   ('保留',   '22C55E'),
    'need':   ('按需',   'EAB308'),
    'clean':  ('可清理', 'EF4444'),
    'careful':('勿乱删', '3B82F6'),
    'unknown':('待确认', 'A855F7'),
}
HDR_FILL = PatternFill('solid', fgColor='1F2937')
HDR_FONT = Font(color='FFFFFF', bold=True, size=11)
thin = Side(style='thin', color='D1D5DB')
BORDER = Border(left=thin, right=thin, top=thin, bottom=thin)

wb = Workbook()

# ---------- Sheet1: 全部清单 ----------
ws = wb.active
ws.title = '全部清单'
cols = ['我的决定', '软件名称', '类别', '用途说明', 'AI建议', '版本', '大小', '来源']
ws.append(cols)
for c in range(1, len(cols) + 1):
    cell = ws.cell(row=1, column=c)
    cell.fill = HDR_FILL; cell.font = HDR_FONT
    cell.alignment = Alignment(horizontal='center', vertical='center')
    cell.border = BORDER

rows = []
for it in BR.main_items:
    rows.append([it['name'], it['cat'], it['desc'], it['advice'], it['ver'], it['size'], '注册表'])
for gname, g in BR.grouped.items():
    rows.append([gname, '组件组', g['desc'], g['advice'], f"{g['count']} 个子项", g['size'], '注册表(合并)'])
for it in BR.store_items:
    rows.append([it['name'], it['cat'], it['desc'], it['advice'], it['ver'], 0, '微软商店'])
for it in BR.extension_items:
    rows.append([it['name'], '系统扩展', '媒体编解码/运行组件（系统级）', 'careful', it['ver'], 0, '微软商店'])

for r in rows:
    ws.append([''] + r)

# 样式：建议列着色 + 我的决定列加下拉
for i in range(2, ws.max_row + 1):
    advice = ws.cell(row=i, column=5).value
    label, color = ADV.get(advice, ('?', '9CA3AF'))
    ws.cell(row=i, column=5).value = label
    ac = ws.cell(row=i, column=5)
    ac.fill = PatternFill('solid', fgColor=color)
    ac.font = Font(color='FFFFFF', bold=True)
    ac.alignment = Alignment(horizontal='center')
    # 大小格式
    sz = ws.cell(row=i, column=7)
    v = sz.value
    sz.value = f'{v/1024:.1f} GB' if v and v >= 1024 else (f'{v:.0f} MB' if v and v >= 1 else '-')
    sz.alignment = Alignment(horizontal='right')
    # 名称加粗
    ws.cell(row=i, column=2).font = Font(bold=True)
    for c in range(1, 9):
        cell = ws.cell(row=i, column=c)
        cell.border = BORDER
        if c in (1, 5, 7):
            cell.alignment = Alignment(horizontal='center', vertical='center')

dv = DataValidation(type='list', formula1='"保留,删除,待定,清理前确认"', allow_blank=True)
dv.error = '请从下拉列表选择'; dv.errorTitle = '无效输入'
ws.add_data_validation(dv)
dv.add(f'A2:A{ws.max_row}')

widths = [12, 34, 12, 52, 10, 18, 10, 14]
for i, w in enumerate(widths, 1):
    ws.column_dimensions[get_column_letter(i)].width = w
ws.freeze_panes = 'A2'
ws.auto_filter.ref = f'A1:H{ws.max_row}'

# ---------- Sheet2: 清理建议 ----------
ws2 = wb.create_sheet('清理建议清单')
ws2.append(['软件名称', '类型', '用途说明', '建议', '大小', '我的决定'])
for c in range(1, 7):
    cell = ws2.cell(row=1, column=c)
    cell.fill = HDR_FILL; cell.font = HDR_FONT
    cell.alignment = Alignment(horizontal='center', vertical='center')
    cell.border = BORDER

clean_rows = []
for it in BR.main_items:
    if it['advice'] in ('clean', 'unknown'):
        clean_rows.append([it['name'], '独立软件', it['desc'], ADV[it['advice']][0], it['size']])
for gname, g in BR.grouped.items():
    if g['advice'] == 'clean':
        clean_rows.append([gname, '组件组', g['desc'], '可清理', g['size']])

for r in clean_rows:
    ws2.append(r + [''])
for i in range(2, ws2.max_row + 1):
    ws2.cell(row=i, column=2).font = Font(bold=True)
    adv = ws2.cell(row=i, column=4).value
    color = ADV.get('clean', ('', 'EF4444'))[1] if adv == '可清理' else 'A855F7'
    ac = ws2.cell(row=i, column=4)
    ac.fill = PatternFill('solid', fgColor=color)
    ac.font = Font(color='FFFFFF', bold=True)
    ac.alignment = Alignment(horizontal='center')
    sz = ws2.cell(row=i, column=5)
    v = sz.value
    sz.value = f'{v/1024:.1f} GB' if v and v >= 1024 else (f'{v:.0f} MB' if v and v >= 1 else '-')
    for c in range(1, 7):
        ws2.cell(row=i, column=c).border = BORDER
dv2 = DataValidation(type='list', formula1='"保留,删除,待定,清理前确认"', allow_blank=True)
ws2.add_data_validation(dv2)
dv2.add(f'F2:F{ws2.max_row}')
for i, w in enumerate([34, 12, 52, 10, 10, 12], 1):
    ws2.column_dimensions[get_column_letter(i)].width = w
ws2.freeze_panes = 'A2'
ws2.auto_filter.ref = f'A1:F{ws2.max_row}'

# ---------- Sheet3: 统计 ----------
ws3 = wb.create_sheet('统计')
from collections import Counter
cat_count = Counter(it['cat'] for it in BR.main_items)
ws3.append(['类别', '软件数'])
for c in range(1, 3):
    cell = ws3.cell(row=1, column=c)
    cell.fill = HDR_FILL; cell.font = HDR_FONT; cell.border = BORDER
for i, (k, v) in enumerate(cat_count.most_common(), 2):
    ws3.cell(row=i, column=1, value=k)
    ws3.cell(row=i, column=2, value=v)
    ws3.cell(row=i, column=1).border = BORDER
    ws3.cell(row=i, column=2).border = BORDER
ws3.column_dimensions['A'].width = 20
ws3.column_dimensions['B'].width = 10

out = r'<LOCAL_USER_HOME>\WorkBuddy\2026-08-24-20-55-17\outputs\软件清点清单.xlsx'
wb.save(out)
print('OK ->', out)
print(f'全部清单行: {ws.max_row - 1}, 清理建议行: {ws2.max_row - 1}, 类别数: {len(cat_count)}')

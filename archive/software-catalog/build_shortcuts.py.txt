def find_exe(name, cat):
    """按名称找 exe：优先精确，再模糊"""
    n = name.lower()
    for k, p in EXE.items():
        if k.lower() in n and p and os.path.isfile(p):
            return p
    return None

# 构建计划
plan = []
skipped = []
for it in BR.main_items:
    name = it['name']
    if excluded(name):
        continue
    # 特别处理：可可影视等未排除但无路径
    exe = find_exe(name, it['cat'])
    folder = FOLDER_MAP.get(it['cat'], '10-系统工具')
    if exe:
        plan.append({'name': name, 'cat': it['cat'], 'folder': folder, 'exe': exe})
    else:
        skipped.append({'name': name, 'cat': it['cat'], 'folder': folder, 'reason': '未找到可执行文件'})

print(f'将创建快捷方式: {len(plan)} 个')
print(f'跳过(未找到exe): {len(skipped)} 个')
for s in skipped:
    print(f'  ⚠ {s["name"]} [{s["cat"]}] - {s["reason"]}')
print('--- 分类统计 ---')
from collections import Counter
c = Counter(p['folder'] for p in plan)
for k, v in sorted(c.items()):
    print(f'  {k}: {v}')

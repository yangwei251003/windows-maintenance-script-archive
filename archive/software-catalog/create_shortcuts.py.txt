def make_lnk(exe, out_lnk, workdir=None, desc=''):
    """克隆模板 lnk 并修改目标路径，生成完整有效的快捷方式"""
    pylnk3.DEFAULT_CHARSET = 'utf-8'  # 支持中文路径
    with open(TEMPLATE, 'rb') as f:
        sc = pylnk3.parse(f)
    # 强制 Unicode 编码，支持中文路径
    sc.link_flags.IsUnicode = True
    li = sc.link_info
    li.local_base_path = exe
    li.make_path = workdir or os.path.dirname(exe)
    if desc:
        sc.description = desc
    sc.work_dir = workdir or os.path.dirname(exe)
    sc.icon = exe
    sc.icon_index = 0
    sc.save(out_lnk)
    return os.path.isfile(out_lnk) and os.path.getsize(out_lnk) > 500

DESKTOP = os.path.join(os.path.expanduser('~'), 'Desktop')
ROOT = os.path.join(DESKTOP, '我的软件')

# 重建（备份旧文件夹）
if os.path.isdir(ROOT):
    backup = os.path.join(DESKTOP, '我的软件_旧版备份')
    if not os.path.isdir(backup):
        try:
            os.rename(ROOT, backup)
            print('旧文件夹已备份为: 我的软件_旧版备份')
        except Exception:
            pass
os.makedirs(ROOT, exist_ok=True)
for folder in sorted(set(v for v in FOLDER_MAP.values() if v)):
    os.makedirs(os.path.join(ROOT, folder), exist_ok=True)

created, skipped = [], []

for it in BR.main_items:
    name = it['name']
    if excluded(name):
        continue
    folder = FOLDER_MAP.get(it['cat'])
    if not folder:
        skipped.append({'name': name, 'reason': '系统组件不建快捷方式'})
        continue

    # 找匹配的 exe 条目
    exe = None
    out_label = None
    for k, (label, p) in APP_EXE.items():
        if k.lower() in name.lower():
            if p:
                exe, out_label = p, label
            break
    if not exe:
        skipped.append({'name': name, 'reason': '未找到可执行文件'})
        continue

    out_lnk = os.path.join(ROOT, folder, out_label + '.lnk')
    if make_lnk(exe, out_lnk, os.path.dirname(exe), out_label):
        created.append({'name': name, 'folder': folder, 'exe': exe})
    else:
        skipped.append({'name': name, 'reason': f'创建失败: {exe}'})

print(f'✅ 创建快捷方式: {len(created)}')
print(f'⚠️ 跳过: {len(skipped)}')
print('--- 跳过明细 ---')
for s in skipped:
    print(f"  {s['name']} - {s['reason']}")
print('--- 分类统计 ---')
from collections import Counter
c = Counter(x['folder'] for x in created)
for k, v in sorted(c.items()):
    print(f'  {k}: {v}')

with open(os.path.join(os.path.dirname(os.path.abspath(__file__)), 'result.json'), 'w', encoding='utf-8') as f:
    json.dump({'created': created, 'skipped': skipped}, f, ensure_ascii=False, indent=1)

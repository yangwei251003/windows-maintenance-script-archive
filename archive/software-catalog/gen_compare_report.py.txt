# -*- coding: utf-8 -*-
"""生成软件对比总结报告（HTML）"""
import json, os, html

TMP = r'<LOCAL_USER_HOME>\AppData\Local\Temp'
with open(os.path.join(TMP, 'installed_software.json'), encoding='utf-8') as f:
    old = json.load(f)
with open(os.path.join(TMP, 'current_software.json'), encoding='utf-8') as f:
    cur = json.load(f)

old_map = {it['Name'].strip().lower(): it for it in old}
cur_map = {it['Name'].strip().lower(): it for it in cur}

gone_names = sorted(set(old_map) - set(cur_map))

# 分类消失的软件（按原始报告的类别）
import sys
sys.path.insert(0, os.path.dirname(os.path.abspath(__file__)))
import build_report as BR

def get_cat(name):
    for kw, (cat, desc, advice) in BR.KB:
        if kw.lower() in name.lower():
            return cat
    return '其他'

# 分组
categories = {}
for n in gone_names:
    # 找到原始名称
    orig = old_map[n]['Name']
    cat = get_cat(orig)
    categories.setdefault(cat, []).append(orig)

CAT_COLORS = {
    '安全防护': '#ef4444', 'AI 工具': '#8b5cf6', '编程开发': '#06b6d4',
    '影音娱乐': '#e879f9', '下载工具': '#0ea5e9', '办公软件': '#3b82f6',
    '设计剪辑': '#ec4899', '字体工具': '#c084fc', '图像工具': '#f472b6',
    '教学仿真': '#84cc16', '预装软件': '#f43f5e', '安卓模拟': '#d946ef',
    'PDF 工具': '#60a5fa', '翻译工具': '#7dd3fc', '其他': '#9ca3af',
}

def esc(s):
    return html.escape(str(s))

cards = ''
for cat, items in sorted(categories.items()):
    c = CAT_COLORS.get(cat, '#9ca3af')
    lis = ''.join(f'<li>{esc(i)}</li>' for i in sorted(items))
    cards += f'''
    <div class="card">
      <div class="card-head">
        <span class="cat-name" style="color:{c}">{esc(cat)}</span>
        <span class="count">{len(items)} 个</span>
      </div>
      <ul>{lis}</ul>
    </div>'''

total_old = len(old)
total_cur = len(cur)
deleted = len(gone_names)

.card {{ background:#161b26; border:1px solid #232a3b; border-radius:12px; padding:14px 16px; }}
.card-head {{ display:flex; justify-content:space-between; align-items:center; margin-bottom:10px; border-bottom:1px solid #232a3b; padding-bottom:8px; }}
.cat-name {{ font-weight:700; font-size:14px; }}
.count {{ font-size:12px; color:#ef4444; background:#ef44441a; padding:2px 8px; border-radius:999px; }}

<div class="stats">

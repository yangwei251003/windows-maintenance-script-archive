# 系统工具兜底
SYS_FOLDER = '10-系统工具'

def classify(name):
    n = name.lower()
    for kws, folder in CAT_MAP:
        for kw in kws:
            if kw.lower() in n:
                return folder
    return SYS_FOLDER

# ---------- 构建快捷方式索引：名称 -> 目标 ----------
# 排除明显非主程序的快捷方式
BAD_TARGETS = ['c:\\users\\', '%windir%', '%programfiles', 'uninstall', '卸载', 'unins', 'msiexec',
               'explorer.exe', 'control.exe', 'cmd.exe', '.chm', '.html', '.url', 'readme',
               'release notes', 'help', 'manuals', 'documentation', 'manual', '迁移', '配置',
               'language', '语言首选项', '遥测', 'spreadsheet compare', 'bonjour', 'virtual',
               'drivers', 'driver', '检查升级', 'checkupdate', 'strong', '强力卸载']

def link_target_for(name):
    """在开始菜单快捷方式中找匹配的 exe"""
    n = name.lower()
    best = None
    for l in links:
        ln = l['Name'].lower()
        t = (l['Target'] or '').lower()
        if not t or any(b in t for b in BAD_TARGETS):
            continue
        # 名称匹配：快捷方式名包含软件名关键词，或软件名包含快捷方式名
        if n in ln or ln in n:
            # 取最长的名称匹配（更精确）
            if best is None or len(ln) > len(best[0]):
                best = (ln, l['Target'], l['Args'])
    return best

# ---------- 从安装位置递归找主 exe ----------
def find_exe_in_dir(d, name):
    if not d or not os.path.isdir(d):
        return None
    # 1) 找与名称相近的 exe
    n = name.lower()
    exes = []
    for root, dirs, files in os.walk(d):
        # 限制深度，避免太慢
        depth = root[len(d):].count(os.sep)
        if depth > 3:
            dirs[:] = []
            continue
        for f in files:
            if f.lower().endswith('.exe'):
                exes.append(os.path.join(root, f))
        if len(exes) > 200:
            break
    if not exes:
        return None
    # 优先匹配名称词
    for exe in exes:
        base = os.path.splitext(os.path.basename(exe))[0].lower()
        if n and n.split()[0] in base:
            return exe
    # 回退：排除 unins/卸载，取第一个
    for exe in exes:
        b = os.path.basename(exe).lower()
        if 'unins' in b or 'uninstall' in b or b.startswith('unins'):
            continue
        return exe
    return None

# ---------- 主流程 ----------
results = []
for item in sw:
    name = item['Name'].strip()
    loc = item.get('InstallLocation') or ''
    ver = item.get('Version') or ''
    if is_excluded(name):
        continue
    folder = classify(name)
    target = None
    src = ''
    # 1) 开始菜单快捷方式
    hit = link_target_for(name)
    if hit and os.path.isfile(hit[1]):
        target = hit[1]
        src = '开始菜单'
    # 2) 已知路径
    if not target:
        for k, p in KNOWN_PATHS.items():
            if k in name.lower() and p and os.path.isfile(p):
                target = p
                src = '已知路径'
                break
    # 3) 安装位置递归
    if not target and loc:
        found = find_exe_in_dir(loc, name)
        if found:
            target = found
            src = '安装目录'
    # 4) 常见用户目录
    if not target:
        common = [
            os.path.join(os.environ.get('LOCALAPPDATA', r'<LOCAL_USER_HOME>\AppData\Local'), 'Programs'),
            r'C:\Program Files', r'C:\Program Files (x86)',
        ]
        n0 = re.split(r'[\s(（]+', name)[0]
        for base in common:
            pat = os.path.join(base, f'*{n0}*')
            for d in glob.glob(pat)[:5]:
                found = find_exe_in_dir(d, name)
                if found:
                    target = found
                    src = '常见目录'
                    break
            if target:
                break
    results.append({
        'name': name, 'ver': ver, 'folder': folder,
        'target': target, 'src': src or '未找到',
    })

# 输出规划 JSON
out_plan = os.path.join(os.path.dirname(os.path.abspath(__file__)), 'plan.json')
with open(out_plan, 'w', encoding='utf-8') as f:
    json.dump(results, f, ensure_ascii=False, indent=1)

foreach ($t in $targets) {
  $line = "--- $($t.Name) ---"
  Add-Content -Path $log -Value $line
  if (-not (Test-Path $t.FilePath)) {
    Add-Content -Path $log -Value "  跳过: 文件不存在 $($t.FilePath)"
    continue
  }
  try {
    $p = Start-Process -FilePath $t.FilePath -ArgumentList $t.Args -PassThru
    $exited = $p.WaitForExit(60000)
    if ($exited) {
      Add-Content -Path $log -Value "  完成: ExitCode=$($p.ExitCode)"
    } else {
      Add-Content -Path $log -Value "  超时: 60s 内未结束(可能卡在界面，未杀进程)"
    }
  } catch {
    Add-Content -Path $log -Value "  异常: $($_.Exception.Message)"
  }
  # 进度提示
  Add-Content -Path $prog -Value $t.Name
}
